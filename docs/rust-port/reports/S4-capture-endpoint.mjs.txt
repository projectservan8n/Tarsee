function raw({ method = "POST", urlPath = "/mcp", token = grant.token, host, headers = {}, body, noAccept = false, noContentType = false, chunked, bodyBytes }) {
  if (bodyBytes) body = bodyBytes;
  return new Promise((resolve, reject) => {
    const h = {};
    if (!noContentType && body != null) h["content-type"] = "application/json";
    if (!noAccept) h.accept = "application/json, text/event-stream";
    Object.assign(h, headers);
    if (token) h.authorization = `Bearer ${token}`;
    if (host) h.host = host;
    if (body != null) h["content-length"] = Buffer.byteLength(body);
    const req = http.request({ host: "127.0.0.1", port: endpoint.port, method, path: urlPath, headers: h, agent: false }, (res) => {
      const chunks = [];
      res.on("data", (c) => chunks.push(c));
      res.on("end", () => resolve({ status: res.statusCode, rawHeaders: res.rawHeaders, body: Buffer.concat(chunks).toString("utf8") }));
    });
    req.on("error", (err) => (err.code === "EPIPE" || err.code === "ECONNRESET" ? resolve({ error: err.code }) : reject(err)));
    if (chunked != null) { req.write(chunked.slice(0, 10)); req.end(chunked.slice(10)); }
    else if (body != null) req.end(body); else req.end();
  });
}

let toolsJson = null;
const out = { timezone: config.TIMEZONE, port: "<port>", cases: {} };
const scrub = (r) => {
  if (!r.rawHeaders) return r;
  const headers = [];
  for (let i = 0; i < r.rawHeaders.length; i += 2) {
    const name = r.rawHeaders[i];
    let value = r.rawHeaders[i + 1];
    if (/^date$/i.test(name)) value = "<date>";
    headers.push([name, value]);
  }
  // The temp folder inside a body (a JSON text: its backslashes are escaped once, or twice where a tool's text
  // quotes a path inside JSON), and the tool list (kept once, in its own fixture).
  let body = r.body;
  const esc = JSON.stringify(tmp).slice(1, -1);
  body = body.split(JSON.stringify(esc).slice(1, -1)).join("<tmp>");
  body = body.split(esc).join("<tmp>");
  if (toolsJson && body.includes(toolsJson)) body = body.split(toolsJson).join('"<tools>"');
  return { status: r.status, headers, body };
};
async function cap(name, req) {
  const body = req.bodyBytes ? undefined : req.body === undefined ? undefined : (typeof req.body === "string" ? req.body : JSON.stringify(req.body));
  const r = await raw({ ...req, body });
  out.cases[name] = { request: { method: req.method || "POST", path: req.urlPath || "/mcp", headers: req.headers || {}, body: body ?? null, ...(req.bodyBytes ? { bodyBase64: req.bodyBytes.toString("base64") } : {}), ...(req.chunked ? { chunked: req.chunked } : {}), ...(req.noAccept ? { noAccept: true } : {}), ...(req.noContentType ? { noContentType: true } : {}), ...(req.token === null ? { noToken: true } : {}), ...(typeof req.token === "string" ? { wrongToken: true } : {}), ...(req.host ? { host: req.host } : {}) }, response: scrub(r) };
  return r;
}
