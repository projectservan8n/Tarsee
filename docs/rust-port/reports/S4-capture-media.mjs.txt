/** A block from its description. */
function block(d) {
  if (d.text !== undefined) return { type: "text", text: d.text };
  if ("raw" in d) return d.raw;
  const source = { type: "base64" };
  let type;
  let bytes;
  if (d.pdf) {
    // (Long pages are named by their count, so the capture stays small.)
    const pages = d.pdf.longPages ? long(d.pdf.longPages)
      : d.pdf.longFirstPage ? ["First. " + "A very long page goes on and on here. ".repeat(d.pdf.longFirstPage), "Second"]
      : d.pdf.pages;
    type = "document"; bytes = makePdf({ pages, compress: !!d.pdf.compress }); source.media_type = "application/pdf";
  }
  else if (d.notPdf !== undefined) { type = "document"; bytes = Buffer.from(d.notPdf); source.media_type = "application/pdf"; }
  else if (d.image !== undefined) { type = "image"; bytes = PNG; if (d.image !== null) source.media_type = d.image; }
  else if (d.audio !== undefined) { type = "audio"; bytes = AUDIO; if (d.audio !== null) source.media_type = d.audio; }
  if (d.mediaType !== undefined) { if (d.mediaType === null) delete source.media_type; else source.media_type = d.mediaType; }
  source.data = bytes.toString("base64");
  if (d.sourceName !== undefined) source.name = d.sourceName;
  const b = { type, source };
  if (d.name !== undefined) b.name = d.name;
  if (d.title !== undefined) b.title = d.title;
  return b;
}

const out = [];
let n = 0;
for (const c of cases) {
  const dir = path.join(tmp, `uploads-${++n}`);
  const content = Array.isArray(c.content) ? c.content.map(block) : c.content;
  const ctx = { uploadsDir: dir, now: NOW };
  if (c.ctx.label !== undefined) ctx.label = c.ctx.label;
  if (c.ctx.transcript !== undefined) ctx.transcribe = async () => ({ transcript: c.ctx.transcript });
  if (c.ctx.transcribeError !== undefined) ctx.transcribe = async () => { throw new Error(c.ctx.transcribeError); };
  const r = await adaptMedia(content, c.caps, ctx);
  const scrub = (t) => String(t).split(dir + path.sep).join("<dir>/");
  // A long text is kept as its ends, its length (in UTF-16 units) and its SHA-256 (of its UTF-8 bytes).
  const brief = (t) => (t.length <= 1500 ? t : { head: t.slice(0, 500), tail: t.slice(-300), length: t.length, sha256: crypto.createHash("sha256").update(t, "utf8").digest("hex") });
  out.push({
    ...c,
    result: {
      text: brief(scrub(r.text)),
      notes: r.notes.map((t) => brief(scrub(t))),
      media: r.media.map((m) => ({ type: m.type, media_type: m.source.media_type, bytes: Buffer.from(m.source.data, "base64").length })),
      contentIsText: typeof r.content === "string",
      contentBlocks: Array.isArray(r.content) ? r.content.map((b) => b.type) : null,
      saved: r.saved.map((s) => ({ kind: s.kind, name: s.name, size: s.size, path: scrub(s.path) })),
      files: fs.existsSync(dir) ? fs.readdirSync(dir).sort() : [],
    },
  });
}
const text = "[\n" + out.map((c) => JSON.stringify(c)).join(",\n") + "\n]\n";
if (text.includes(tmp)) throw new Error("the temp folder is still in the capture");
fs.writeFileSync(outFile, text);
console.log(out.length, "cases");
fs.rmSync(tmp, { recursive: true, force: true });
process.exit(0);
