/** A long text as its digest and size, so the file stays small: the Rust test hashes its own the same way. */
const digest = (text) => ({ sha256: crypto.createHash("sha256").update(text, "utf8").digest("hex"), bytes: Buffer.byteLength(text) });
const LONG = 1000;
const short = (text) => (typeof text === "string" && text.length > LONG ? digest(text) : text);

function errorOf(err) {
  return { name: err?.constructor?.name, code: err?.code ?? null, message: String(err?.message ?? err) };
}

const builds = [];
for (const provider of providers) {
  for (const [name, target, change] of variants) {
    // The other spellings of an id only once each.
    if (provider !== provider.trim().toLowerCase() && name !== "base") continue;
    const id = provider.trim().toLowerCase();
    const ctx = { ...base(id), ...change };
    for (const [k, v] of Object.entries(change)) if (v === null) delete ctx[k];
    // What the case changes in the base context (null: left out); the Rust test builds the base context itself.
    const input = { provider, target, change: { ...change, ...("developerInstructions" in change ? { developerInstructions: short(change.developerInstructions) } : {}) } };
    try {
      const o = co.buildCodexOptions({ providerId: provider, engine: "codex", ...target }, ctx);
      const { home, ...rest } = o;
      const env = { ...rest.env };
      delete env.CODEX_HOME;
      const config = { ...rest.config, developer_instructions: short(rest.config.developer_instructions) };
      const args = co.configArgs(o.config).map((a) => (a.length > LONG ? digest(a) : a));
      // The flattened arguments for a few of them (the config itself is compared key by key, in order, for all).
      const withArgs = ["base", "model and effort", "attribution", "test hosts", "context", "compact off"].includes(name);
      builds.push({ name, input, out: { ...rest, env, config }, home, codexHome: o.env.CODEX_HOME, ...(withArgs ? { args } : {}) });
    } catch (err) {
      builds.push({ name, input, error: errorOf(err) });
    }
  }
}
