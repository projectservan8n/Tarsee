const plain = (v) => JSON.parse(JSON.stringify(v ?? null));
const stateOf = (s) => ({ threadId: s.threadId, totals: s.totals, done: s.done, toolCount: s.toolCount, label: s.label, providerId: s.providerId });

/** A run: the events in order (an "x-sdk-threw" line is the error finish() sees), then finish(). */
function run(opts, events, finish = {}) {
  const state = cx.createCodexEventState(opts);
  const out = [];
  let thrown = null;
  for (const ev of events) {
    if (ev && ev.type === "x-sdk-threw") { thrown = new Error(ev.message); continue; }
    out.push(...cx.mapCodexEvent(ev, state));
  }
  const fin = { ...finish };
  if (thrown && !("error" in fin)) fin.error = thrown;
  const end = cx.finishCodexEvents(state, fin);
  return { out: plain(out), end: plain(end), state: plain(stateOf(state)) };
}

function finishText(f) {
  const out = { ...f };
  if (f.error instanceof Error) out.error = { message: f.error.message };
  else if (typeof f.error === "string") out.error = { string: f.error };
  return out;
}
